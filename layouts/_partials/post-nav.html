{{- /* 文章上一篇 / 下一篇。参数：页面。 系列文章用系列内的前后篇（prev = 阅读顺序更早，next = 更晚）； 非系列文章按时间：prev = 更早发布，next = 更晚发布。两者都没有时不输出。 */ -}} {{- $prev := false -}} {{- $next := false -}} {{- with partial "funcs/series.html" . -}} {{- $prev = .prev -}} {{- $next = .next -}} {{- else -}} {{- /* 实测（Hugo 0.169）：.PrevInSection 是日期更早的一篇，.NextInSection 是更晚的一篇，与阅读方向一致 */ -}} {{- $prev = .PrevInSection -}} {{- $next = .NextInSection -}} {{- end -}} {{- if or $prev $next }}
{{- with $prev }} {{ i18n "prev" }}{{ .LinkTitle }} {{- end }} {{- with $next }} {{ i18n "next" }}{{ .LinkTitle }} {{- end }}
{{- end }}
