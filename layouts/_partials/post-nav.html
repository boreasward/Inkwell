{{- /* 文章上一篇 / 下一篇。参数：页面。 系列文章用系列内的前后篇（prev = 阅读顺序更早，next = 更晚）； 非系列文章按时间：prev = 更早发布，next = 更晚发布。两者都没有时不输出。 */ -}} {{- $prev := false -}} {{- $next := false -}} {{- with partial "funcs/series.html" . -}} {{- $prev = .prev -}} {{- $next = .next -}} {{- else -}} {{- /* 非系列：只有文章（站点主栏目里的页面，与首页列表同一口径）才有时间上的前后篇， 关于、归档等独立页面不输出。严格按日期升序，prev = 更早，next = 更晚。 */ -}} {{- if in site.MainSections .Type -}} {{- $posts := (where site.RegularPages "Type" "in" site.MainSections).ByDate -}} {{- range $i, $p := $posts -}} {{- if $p.Eq $ -}} {{- if gt $i 0 }}{{ $prev = index $posts (sub $i 1) }}{{ end -}} {{- if lt (add $i 1) (len $posts) }}{{ $next = index $posts (add $i 1) }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if or $prev $next }}

{{- end }}
